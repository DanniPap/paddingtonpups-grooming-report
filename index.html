const GROOMING_SHEET_NAME = 'Grooming'; const GROOMING_SPREADSHEET_ID = '1GMllcpqls8Gn6gSxjwKZoxOhyY7Fyq9wVKFjVSza49s'; const GROOMING_HEADERS = [ 'Date', 'Total Hours', 'Groomers', 'Total Dogs', 'Total Slots', 'No-Shows', 'New Clients', 'Dog Behaviour Score', 'Dog Behaviour', 'Groomer Detail', 'Notes', 'Raw JSON' ]; function groomingReportToRow(report) { const groomerNames = (report.groomers || []).map(g => `${g.name} (${g.hours}h)`).join(', '); const groomerDetail = (report.groomers || []).map(g => { const filled = (g.slots || []).filter(s => s.dogName && s.dogName.trim()); return `${g.name} (${g.hours}h): ${filled.map(s => `${s.dogName} - ${s.service}${s.isLargeXL ? ' [L/XL]' : ''}`).join(', ')}`; }).join(' | '); const behaviourLabels = {1:'Very Difficult',2:'Mostly Difficult',3:'Challenging',4:'Below Average',5:'Average',6:'Above Average',7:'Good',8:'Well Behaved',9:'Excellent',10:'Perfect Angels'}; return [ report.date, report.totalHours, groomerNames, report.totalDogs, report.totalSlots, report.ops?.noShows ?? 0, report.ops?.newClients ?? 0, report.ops?.dogBehaviour ?? 5, behaviourLabels[report.ops?.dogBehaviour ?? 5], groomerDetail, report.notes || '', JSON.stringify(report) ]; } function doGetGrooming(e) { const action = e.parameter.action; if (action === 'saveGrooming') { try { const data = JSON.parse(decodeURIComponent(e.parameter.data)); saveGroomingReport(data.key, data.report); return ContentService.createTextOutput(JSON.stringify({ success: true })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ error: err.message })) .setMimeType(ContentService.MimeType.JSON); } } return getAllGroomingReports(); } function saveGroomingReport(key, report) { const ss = SpreadsheetApp.openById(GROOMING_SPREADSHEET_ID); let sheet = ss.getSheetByName(GROOMING_SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(GROOMING_SHEET_NAME); sheet.appendRow(['_key', ...GROOMING_HEADERS]); } const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (rows[i][0] === key) { sheet.getRange(i + 1, 2, 1, GROOMING_HEADERS.length).setValues([groomingReportToRow(report)]); return; } } sheet.appendRow([key, ...groomingReportToRow(report)]); } function getAllGroomingReports() { const ss = SpreadsheetApp.openById(GROOMING_SPREADSHEET_ID); const sheet = ss.getSheetByName(GROOMING_SHEET_NAME); if (!sheet) { return ContentService.createTextOutput(JSON.stringify([])) .setMimeType(ContentService.MimeType.JSON); } const rows = sheet.getDataRange().getValues(); const reports = []; for (let i = 1; i < rows.length; i++) { if (rows[i][0] && rows[i][12]) { try { const report = JSON.parse(rows[i][12]); reports.push({ ...report, _storageKey: rows[i][0] }); } catch(e) {} } } return ContentService.createTextOutput(JSON.stringify(reports)) .setMimeType(ContentService.MimeType.JSON); }
